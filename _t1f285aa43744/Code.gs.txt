const TZ = 'Asia/Bangkok';
const SHEET_NAME = 'Orders';
const UPLOAD_FOLDER = 'Yograce รูปเมนู (อัปโหลดจากหน้าพนักงาน)';

/* ---------- สถานะร้าน + เมนูที่แก้จากหน้าพนักงาน ---------- */
// STATE = { paused, soldOut:[id], added:[{id,cat,name,en,price,img}], removed:[id], edits:{id:{name,en,price,img}}, isNew:{id:true|false} }
function state_() {
  let s = {};
  try { s = JSON.parse(PropertiesService.getScriptProperties().getProperty('STATE') || '{}'); } catch (e) {}
  return Object.assign({ paused: false, soldOut: [], added: [], removed: [], edits: {}, isNew: {} }, s);
}
function saveState_(s) {
  PropertiesService.getScriptProperties().setProperty('STATE', JSON.stringify(s));
}

// เมนูจริง = เมนูตั้งต้นใน Menu.gs + การแก้ไขจากหน้าพนักงาน
function menu_() {
  const s = state_();
  const images = Object.assign({}, typeof IMAGES !== 'undefined' ? IMAGES : {});
  const menu = MENU.map(c => ({ cat: c.cat, name: c.name, icon: c.icon,
    items: c.items.filter(i => s.removed.indexOf(i.id) < 0).map(i => Object.assign({}, i)) }));
  s.added.forEach(a => {
    const c = menu.find(x => x.cat === a.cat) || menu[0];
    if (s.removed.indexOf(a.id) < 0) c.items.push({ id: a.id, name: a.name, en: a.en || '', price: a.price });
    if (a.img) images[a.id] = a.img;
  });
  menu.forEach(c => c.items.forEach(i => {
    const e = s.edits[i.id];
    if (e) {
      if (e.name) i.name = e.name;
      if (e.en != null) i.en = e.en;
      if (e.price != null) i.price = e.price;
      if (e.img) images[i.id] = e.img;
    }
  }));
  const ids = {}; menu.forEach(c => c.items.forEach(i => ids[i.id] = 1));
  const news = (CONFIG.NEW_ITEMS || []).filter(id => s.isNew[id] !== false);
  Object.keys(s.isNew).forEach(id => { if (s.isNew[id] && news.indexOf(id) < 0) news.push(id); });
  return {
    MENU: menu, IMAGES: images,
    NEW_ITEMS: news.filter(id => ids[id]),
    SOLD_OUT: s.soldOut.filter(id => ids[id]),
    PAUSED: !!s.paused,
  };
}

/* ---------- เวลาเปิดรับ ---------- */
function nowHM_() { return Utilities.formatDate(new Date(), TZ, 'HH:mm'); }
function openStatus_(m) {
  if (m.PAUSED) return { open: false, why: 'ขออภัย ออเดอร์เต็มชั่วคราว กรุณาลองใหม่อีกครั้งภายหลัง' };
  const t = nowHM_();
  if (t < CONFIG.OPEN_TIME || t > CONFIG.CLOSE_TIME)
    return { open: false, why: 'ปิดรับออเดอร์แล้ว เปิดรับทุกวัน ' + CONFIG.OPEN_TIME + '–' + CONFIG.CLOSE_TIME + ' น.' };
  return { open: true };
}

/* ---------- หน้าเว็บสั่งอาหาร ---------- */
function publicData_() {
  const m = menu_();
  const cfg = Object.assign({}, CONFIG, { NEW_ITEMS: m.NEW_ITEMS });
  return { CONFIG: cfg, MENU: m.MENU, TOPPINGS, SWEETNESS, IMAGES: m.IMAGES, SOLD_OUT: m.SOLD_OUT, PAUSED: m.PAUSED };
}
function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const p = (e && e.parameter) || {};
  // ?api=menu → ส่งข้อมูลเมนูเป็น JSON ให้หน้าสั่งที่โฮสต์ไว้ข้างนอก (GitHub Pages)
  if (p.api === 'menu') return json_(publicData_());
  // ?qr=1&a=ยอด&o=เลขออเดอร์ → เปิดหน้า QR อย่างเดียว (ใช้ตอนลูกค้าเปิดนอก LINE เพื่อบันทึกรูป)
  let qrview = null;
  if (p.qr) {
    const a = Number(p.a);
    const o = String(p.o || '').replace(/[^A-Za-z0-9-]/g, '').slice(0, 20);
    if (a > 0 && a < 100000) qrview = { amount: Math.round(a * 100) / 100, orderNo: o };
  }
  const t = HtmlService.createTemplateFromFile('Index');
  t.data = JSON.stringify(Object.assign(publicData_(), { QRVIEW: qrview, EXEC_URL: ScriptApp.getService().getUrl() }));
  return t.evaluate()
    .setTitle('สั่ง Yograce Delivery')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/* ---------- คำนวณราคา (ฝั่งเซิร์ฟเวอร์ ใช้ราคาจากระบบเสมอ) ---------- */
function findItem_(m, id) {
  for (const c of m.MENU) for (const i of c.items) if (i.id === id) return i;
  return null;
}
function findTopping_(id) {
  return TOPPINGS.find(t => t.id === id) || null;
}
function deliveryFee_(cups, dateStr) {
  if (dateStr <= CONFIG.FREE_UNTIL) return 0;
  return cups >= CONFIG.MIN_CUPS_FREE ? 0 : CONFIG.DELIVERY_FEE;
}
function calcOrder_(m, items) {
  const today = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
  let cups = 0, subtotal = 0;
  const lines = [], soldOut = [], missing = [];
  (items || []).forEach(it => {
    const item = findItem_(m, it.id);
    const qty = Math.max(1, Math.min(50, parseInt(it.qty, 10) || 1));
    if (!item) { missing.push(String(it.id)); return; }
    if (m.SOLD_OUT.indexOf(item.id) >= 0) soldOut.push(item.name);
    const tops = (it.toppings || []).map(findTopping_).filter(Boolean);
    const unit = item.price + tops.reduce((s, t) => s + t.price, 0);
    cups += qty;
    subtotal += unit * qty;
    lines.push({ name: item.name, qty, unit, sweet: String(it.sweet || '100%').slice(0, 5),
      toppings: tops.map(t => t.name), note: String(it.note || '').slice(0, 100) });
  });
  const fee = deliveryFee_(cups, today);
  return { lines, cups, subtotal, fee, total: subtotal + fee, soldOut, missing };
}
function readCustomer_(c) {
  c = c || {};
  const name = String(c.name || '').trim().slice(0, 60);
  const phone = String(c.phone || '').replace(/\D/g, '');
  const address = String(c.address || '').trim().slice(0, 300);
  let points = String(c.points || '').replace(/\D/g, '');
  if (!/^0\d{8,9}$/.test(points)) points = '';
  if (!name || !/^0\d{8,9}$/.test(phone) || !address) {
    throw new Error('กรุณากรอกชื่อ เบอร์โทร และที่อยู่ให้ครบ');
  }
  return {
    name, phone, points, address,
    map: String(c.map || '').slice(0, 300),
    distance: c.distance != null && c.distance !== '' ? Number(c.distance).toFixed(1) : '',
    when: String(c.when || 'ส่งทันที').slice(0, 40),
    note: String(c.note || '').slice(0, 200),
  };
}

/* ขั้นที่ 1: เช็กเวลา/ของหมด/ยอดเงิน ก่อนแสดง QR ให้ลูกค้าโอน */
function quoteOrder(payload) {
  readCustomer_(payload.customer);
  const m = menu_();
  const st = openStatus_(m);
  if (!st.open) throw new Error(st.why);
  const o = calcOrder_(m, payload.items);
  if (!o.lines.length) throw new Error('ยังไม่มีรายการในตะกร้า');
  if (o.missing.length) throw new Error('มีเมนูในตะกร้าที่ร้านเอาออกแล้ว กรุณาลบออกจากตะกร้าแล้วลองใหม่');
  if (o.soldOut.length) throw new Error('ขออภัย เมนูนี้หมดแล้ว: ' + o.soldOut.join(', ') + ' — กรุณาลบออกจากตะกร้า');
  return { ok: true, total: o.total, fee: o.fee, cups: o.cups, subtotal: o.subtotal };
}

/* สร้างออเดอร์
   - หน้าสั่งปัจจุบัน (strict): เช็กเวลาเปิด/เมนูหมดก่อน → ส่งเข้ากลุ่ม → หน้าเว็บแสดง QR ให้ลูกค้าโอน
   - หน้าเวอร์ชันที่ลูกค้าโอนก่อนแล้วกดยืนยัน (มี quoted): รับเสมอ แต่แจ้งเตือนพนักงานถ้าผิดปกติ */
function submitOrder(payload) {
  const cust = readCustomer_(payload.customer);
  const m = menu_();
  const o = calcOrder_(m, payload.items);
  if (!o.lines.length) throw new Error('ยังไม่มีรายการในตะกร้า');
  if (payload.strict) {
    const st0 = openStatus_(m);
    if (!st0.open) throw new Error(st0.why);
    if (o.missing.length) throw new Error('มีเมนูในตะกร้าที่ร้านเอาออกแล้ว กรุณาลบออกจากตะกร้าแล้วลองใหม่');
    if (o.soldOut.length) throw new Error('ขออภัย เมนูนี้หมดแล้ว: ' + o.soldOut.join(', ') + ' — กรุณาลบออกจากตะกร้า');
  }

  const warn = [];
  const shown = Number(payload.quoted);
  if (shown && shown !== o.total) warn.push('ยอดที่ลูกค้าเห็นตอนโอน ' + shown + '฿ ไม่ตรงกับยอดระบบ ' + o.total + '฿');
  if (o.soldOut.length) warn.push('เมนูหมด: ' + o.soldOut.join(', '));
  if (o.missing.length) warn.push('มีเมนูที่ถูกเอาออกจากร้านแล้วในตะกร้า');
  const st = openStatus_(m);
  if (!st.open) warn.push('ยืนยันนอกเวลา/ช่วงปิดรับชั่วคราว');

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let orderNo;
  try {
    const props = PropertiesService.getScriptProperties();
    const day = Utilities.formatDate(new Date(), TZ, 'yyMMdd');
    const key = 'SEQ_' + day;
    const seq = (parseInt(props.getProperty(key), 10) || 0) + 1;
    props.setProperty(key, String(seq));
    orderNo = 'YG' + day + '-' + ('00' + seq).slice(-3);
  } finally {
    lock.releaseLock();
  }

  // หน้าเว็บเวอร์ชันเก่า (ยังค้างในมือถือ) ไม่มีขั้นโอนก่อน → ใช้วิธีชำระที่ลูกค้าเลือก
  const pay = payload.strict ? 'โอน (รอลูกค้าโอน)'
    : payload.qrFirst ? 'โอน (ลูกค้าได้ QR ก่อนกดสั่ง)'
    : payload.quoted != null ? 'โอน (ลูกค้ากดว่าโอนแล้ว)'
    : String((payload.customer || {}).pay || 'เงินสดปลายทาง').slice(0, 30);
  const order = Object.assign({ orderNo, pay, warn }, cust, o);
  saveToSheet_(order);
  // ลูกค้าที่สั่งผ่าน LIFF ใน LINE → เก็บสรุปไว้ให้บอทตอบกลับในแชท (จับคู่ด้วย LINE userId)
  const uid = String(payload.lineUid || '');
  if (/^U[0-9a-f]{32}$/.test(uid)) {
    CacheService.getScriptCache().put('OS_' + orderNo, JSON.stringify({ uid, text: buildCustomerReply_(order) }), 21600);
  }
  const text = buildMessage_(order);
  let pushed = true;
  try { pushToGroup_(text); } catch (err) { pushed = false; console.error(err); }

  return { ok: true, orderNo, total: o.total, fee: o.fee, cups: o.cups, pushed, summary: buildCustomerText_(order) };
}

/* ---------- ข้อความเข้ากลุ่มพนักงาน ---------- */
function buildMessage_(o) {
  const L = [];
  L.push('🧾 ออเดอร์ใหม่ ' + o.orderNo);
  L.push('🕒 ' + Utilities.formatDate(new Date(), TZ, 'dd/MM HH:mm') + ' | ' + o.when);
  L.push('━━━━━━━━━━');
  o.lines.forEach((l, i) => {
    L.push((i + 1) + ') ' + l.name + ' x' + l.qty + '  ' + (l.unit * l.qty) + '฿');
    const extra = ['หวาน ' + l.sweet];
    if (l.toppings.length) extra.push('+' + l.toppings.join(', +'));
    L.push('    ' + extra.join(' | '));
    if (l.note) L.push('    📝 ' + l.note);
  });
  L.push('━━━━━━━━━━');
  L.push('รวม ' + o.cups + ' แก้ว  ค่าอาหาร ' + o.subtotal + '฿');
  L.push('ค่าส่ง ' + (o.fee ? o.fee + '฿' : 'ฟรี'));
  if (/^โอน \(/.test(o.pay)) {
    L.push('💰 ยอดโอน ' + o.total + '฿');
    L.push(/กดว่าโอนแล้ว/.test(o.pay) ? '⏳ ลูกค้ากดยืนยันว่าโอนแล้ว — เช็กยอดในถุงเงินก่อนทำ'
      : /QR ก่อนกดสั่ง/.test(o.pay) ? '⏳ ลูกค้าได้ QR ก่อนกดสั่ง — เช็กยอดในถุงเงินก่อนทำ'
      : '⏳ ลูกค้าจะโอนตาม QR — เช็กยอดในถุงเงินก่อนทำ');
  } else {
    L.push('💰 ยอดเก็บ ' + o.total + '฿ (' + o.pay + ')');
  }
  (o.warn || []).forEach(w => L.push('⚠️ ' + w));
  L.push('━━━━━━━━━━');
  L.push('👤 ' + o.name + '  📞 ' + o.phone);
  if (o.points) L.push('⭐ สะสมแต้ม: ' + o.points);
  L.push('📍 ' + o.address);
  if (o.map) L.push('🗺️ ' + o.map + (o.distance ? ' (~' + o.distance + ' กม.)' : ''));
  if (o.note) L.push('📝 ' + o.note);
  return L.join('\n');
}

function buildCustomerReply_(o) {
  const L = ['✅ ร้านได้รับออเดอร์แล้ว', 'เลขออเดอร์ ' + o.orderNo, '━━━━━━━━━━'];
  o.lines.forEach(l => {
    L.push('• ' + l.name + ' x' + l.qty + '  ' + (l.unit * l.qty) + '฿');
    L.push('   หวาน ' + l.sweet + (l.toppings.length ? ' · +' + l.toppings.join(', +') : '') + (l.note ? ' · ' + l.note : ''));
  });
  L.push('━━━━━━━━━━');
  L.push('รวม ' + o.cups + ' แก้ว · ค่าส่ง ' + (o.fee ? o.fee + ' บาท' : 'ฟรี'));
  L.push('💰 ยอดโอน ' + o.total + ' บาท');
  L.push('📍 ' + o.address);
  L.push('🕒 ' + o.when);
  L.push('━━━━━━━━━━');
  L.push('ร้านจะเริ่มทำเมื่อยอดโอนเข้า ไรเดอร์จะโทรหาเมื่อใกล้ถึง');
  L.push('ขอบคุณที่สั่ง Yograce 💚');
  return L.join('\n');
}

function buildCustomerText_(o) {
  const L = ['สั่ง Yograce ' + o.orderNo];
  o.lines.forEach(l => L.push('- ' + l.name + ' x' + l.qty + ' (หวาน ' + l.sweet +
    (l.toppings.length ? ', +' + l.toppings.join(', +') : '') + ')'));
  L.push('ยอดรวม ' + o.total + ' บาท (ค่าส่ง ' + (o.fee ? o.fee + ' บาท' : 'ฟรี') + ')');
  L.push('ชำระ: ' + (/^โอน \(/.test(o.pay) ? 'โอนพร้อมเพย์' : o.pay));
  return L.join('\n');
}

/* ---------- LINE Messaging API ---------- */
function lineApi_(path, body) {
  const token = PropertiesService.getScriptProperties().getProperty('LINE_TOKEN');
  if (!token) throw new Error('ยังไม่ได้ตั้งค่า LINE_TOKEN');
  const res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/' + path, {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify(body),
    muteHttpExceptions: true,
  });
  if (res.getResponseCode() !== 200) throw new Error('LINE ' + res.getResponseCode() + ': ' + res.getContentText());
}

function pushToGroup_(text) {
  const gid = PropertiesService.getScriptProperties().getProperty('GROUP_ID');
  if (!gid) throw new Error('ยังไม่มี GROUP_ID — เชิญบอทเข้ากลุ่มก่อน');
  lineApi_('push', { to: gid, messages: [{ type: 'text', text: text.slice(0, 4900) }] });
}

/* doPost: ออเดอร์/เช็กยอดจากหน้าเว็บ, คำสั่งจากหน้าพนักงาน, และ Webhook จาก LINE
   (เชิญบอทเข้ากลุ่ม → บันทึก GROUP_ID อัตโนมัติ หรือพิมพ์ "ผูกกลุ่ม" ในกลุ่ม) */
function doPost(e) {
  let body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) {}
  const run = fn => { try { return json_(fn(body)); } catch (err) { return json_({ ok: false, error: String((err && err.message) || err) }); } };
  if (body.action === 'quote') return run(quoteOrder);
  if (body.action === 'order') return run(submitOrder);
  if (body.action === 'admin') return run(admin_);
  (body.events || []).forEach(ev => {
    const src = ev.source || {};
    // แชท 1:1 กับลูกค้า: ข้อความ "สั่งออเดอร์ YG..." ที่ส่งจากหน้าสั่ง (LIFF) → ตอบกลับสรุปออเดอร์ (ข้อความตอบกลับไม่นับโควตา)
    if (src.type === 'user' && ev.type === 'message' && ev.message && ev.message.type === 'text') {
      const m = /YG\d{6}-\d{3,}/.exec(ev.message.text || '');
      if (!m) return;
      let saved = null;
      try { saved = JSON.parse(CacheService.getScriptCache().get('OS_' + m[0]) || 'null'); } catch (err) {}
      if (!saved || saved.uid !== src.userId) return;
      try {
        lineApi_('reply', { replyToken: ev.replyToken, messages: [{ type: 'text', text: saved.text.slice(0, 4900) }] });
      } catch (err) { console.error(err); }
      return;
    }
    if (src.type !== 'group') return;
    const isJoin = ev.type === 'join';
    const isCmd = ev.type === 'message' && ev.message && ev.message.type === 'text' &&
      ev.message.text.trim() === 'ผูกกลุ่ม';
    if (isJoin || isCmd) {
      PropertiesService.getScriptProperties().setProperty('GROUP_ID', src.groupId);
      try {
        lineApi_('reply', { replyToken: ev.replyToken,
          messages: [{ type: 'text', text: '✅ เชื่อมกลุ่มนี้กับระบบรับออเดอร์ Yograce แล้ว' }] });
      } catch (err) { console.error(err); }
    }
  });
  return ContentService.createTextOutput('OK');
}

/* ---------- หน้าพนักงาน (yograceyogurt.com/staff) ---------- */
function checkPin_(pin) {
  const cache = CacheService.getScriptCache();
  const fails = parseInt(cache.get('PIN_FAILS') || '0', 10);
  if (fails >= 10) throw new Error('ใส่ PIN ผิดหลายครั้ง กรุณารอ 15 นาที');
  const real = PropertiesService.getScriptProperties().getProperty('ADMIN_PIN');
  if (!real) throw new Error('ยังไม่ได้ตั้ง PIN ในระบบ');
  if (String(pin || '') !== real) {
    cache.put('PIN_FAILS', String(fails + 1), 900);
    throw new Error('PIN ไม่ถูกต้อง');
  }
}
function cleanText_(v, n) { return String(v == null ? '' : v).replace(/\s+/g, ' ').trim().slice(0, n); }
function cleanPrice_(v) {
  const p = Math.round(Number(v));
  if (!(p > 0 && p < 10000)) throw new Error('ราคาไม่ถูกต้อง');
  return p;
}
function uploadImage_(dataUrl, name) {
  const m = /^data:(image\/(?:jpeg|png|webp));base64,(.+)$/.exec(String(dataUrl || ''));
  if (!m) throw new Error('ไฟล์รูปไม่ถูกต้อง');
  const bytes = Utilities.base64Decode(m[2]);
  if (bytes.length > 3 * 1024 * 1024) throw new Error('รูปใหญ่เกินไป');
  const folder = uploadFolder_();
  const file = folder.createFile(Utilities.newBlob(bytes, m[1], name + '.' + m[1].split('/')[1]));
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file.getId();
}
function uploadFolder_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('UPLOAD_FOLDER_ID');
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  const f = DriveApp.createFolder(UPLOAD_FOLDER);
  props.setProperty('UPLOAD_FOLDER_ID', f.getId());
  return f;
}

function admin_(b) {
  checkPin_(b.pin);
  const op = b.op;
  if (op === 'get') return adminView_();
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const s = state_();
    const m = menu_();
    const exists = id => !!findItem_(m, id);
    if (op === 'pause') {
      s.paused = !!b.paused;
    } else if (op === 'soldout') {
      if (!exists(b.id)) throw new Error('ไม่พบเมนูนี้');
      s.soldOut = s.soldOut.filter(x => x !== b.id);
      if (b.soldOut) s.soldOut.push(b.id);
    } else if (op === 'add') {
      const cat = MENU.find(c => c.cat === b.cat) ? b.cat : MENU[0].cat;
      const name = cleanText_(b.name, 60);
      if (!name) throw new Error('กรุณาใส่ชื่อเมนู');
      const id = 'n' + Date.now().toString(36);
      const item = { id, cat, name, en: cleanText_(b.en, 60), price: cleanPrice_(b.price) };
      if (b.image) item.img = uploadImage_(b.image, id);
      s.added.push(item);
      if (b.isNew) s.isNew[id] = true;
    } else if (op === 'edit') {
      if (!exists(b.id)) throw new Error('ไม่พบเมนูนี้');
      const name = cleanText_(b.name, 60);
      if (!name) throw new Error('กรุณาใส่ชื่อเมนู');
      const patch = { name, en: cleanText_(b.en, 60), price: cleanPrice_(b.price) };
      if (b.image) patch.img = uploadImage_(b.image, b.id + '-' + Date.now().toString(36));
      const added = s.added.find(a => a.id === b.id);
      if (added) {
        Object.assign(added, patch);
        if (b.cat && MENU.find(c => c.cat === b.cat)) added.cat = b.cat;
      } else {
        s.edits[b.id] = Object.assign({}, s.edits[b.id] || {}, patch);
      }
      s.isNew[b.id] = !!b.isNew;
    } else if (op === 'remove') {
      if (!exists(b.id)) throw new Error('ไม่พบเมนูนี้');
      if (s.added.find(a => a.id === b.id)) s.added = s.added.filter(a => a.id !== b.id);
      else s.removed.push(b.id);
      s.soldOut = s.soldOut.filter(x => x !== b.id);
      delete s.isNew[b.id];
      delete s.edits[b.id];
    } else {
      throw new Error('คำสั่งไม่ถูกต้อง');
    }
    saveState_(s);
  } finally {
    lock.releaseLock();
  }
  return adminView_();
}
function adminView_() {
  const m = menu_();
  const st = openStatus_(Object.assign({}, m, { PAUSED: false }));
  const added = {}; state_().added.forEach(a => added[a.id] = 1);
  return {
    ok: true,
    paused: m.PAUSED, inHours: st.open, hours: CONFIG.OPEN_TIME + '–' + CONFIG.CLOSE_TIME,
    cats: MENU.map(c => ({ cat: c.cat, name: c.name, icon: c.icon })),
    menu: m.MENU.map(c => ({ cat: c.cat, name: c.name, icon: c.icon, items: c.items.map(i => ({
      id: i.id, name: i.name, en: i.en, price: i.price, img: m.IMAGES[i.id] || '',
      soldOut: m.SOLD_OUT.indexOf(i.id) >= 0, isNew: m.NEW_ITEMS.indexOf(i.id) >= 0, added: !!added[i.id],
    })) })),
  };
}

/* ---------- Google Sheet ---------- */
function saveToSheet_(o) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(['เวลา', 'เลขออเดอร์', 'ชื่อ', 'เบอร์', 'เบอร์สะสมแต้ม', 'ที่อยู่', 'แผนที่', 'ระยะ(กม.)',
      'เวลาที่ต้องการ', 'ชำระเงิน', 'รายการ', 'จำนวนแก้ว', 'ค่าอาหาร', 'ค่าส่ง', 'ยอดรวม', 'หมายเหตุ', 'สถานะ']);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, 17).setFontWeight('bold').setBackground('#DDEFE6');
  }
  const items = o.lines.map(l => l.name + ' x' + l.qty + ' [' + l.sweet +
    (l.toppings.length ? ' +' + l.toppings.join(' +') : '') + ']' + (l.note ? ' (' + l.note + ')' : '')).join('\n');
  const note = [o.note].concat((o.warn || []).map(w => '⚠️ ' + w)).filter(Boolean).join('\n');
  sh.appendRow([new Date(), o.orderNo, o.name, "'" + o.phone, o.points ? "'" + o.points : '', o.address, o.map, o.distance,
    o.when, o.pay, items, o.cups, o.subtotal, o.fee, o.total, note, 'ใหม่']);
}
